public abstract class LivingEntity extends Entity implements Attackable, WaypointTransmitter {
   protected void tryAddFrost() {
      if (!this.getBlockStateOnLegacy().isAir()) {
         int ticksFrozen = this.getTicksFrozen();
         if (ticksFrozen > 0) {
            AttributeInstance speed = this.getAttribute(Attributes.MOVEMENT_SPEED);
            if (speed == null) {
               return;
            }

            float slowAmount = -0.05F * this.getPercentFrozen();
            speed.addTransientModifier(new AttributeModifier(SPEED_MODIFIER_POWDER_SNOW_ID, slowAmount, Operation.ADD_VALUE));
         }
      }
   }
}
